function validateCutlineRound(input) {
  const totalRounds = parseInt(document.querySelector('.row-input').value || 2);
  const val = parseInt(input.value);
  if (isNaN(val) || val < 1 || val >= totalRounds) {
    input.value = Math.round(totalRounds / 2);
  }
}

function selectTag(el){
  const siblings = el.parentNode.querySelectorAll('.tag');
  siblings.forEach(t=>t.classList.remove('active'));
  el.classList.add('active');
}

  if(cutlineRoundInput) {
    cutlineRoundInput.value = Math.round(count / 2);
  }
  
  box.innerHTML = "";
  const options = ["四人四球比洞", "四人两球比洞", "最佳球位比洞", "个人比洞"];

  for(let i=0;i<count;i++){

function toggleGlobalCutline(isYes) {
  const yesBtn = document.getElementById('cutline-yes');

let activeTimeButton = null;
let activeRoundIndex = null;
let currentSelectedYear = new Date().getFullYear();

function populateWheel(id, values, labelSuffix = '') {
  const wheel = document.getElementById(id);
  let html = '<div class="wheel-spacer"></div>';
  values.forEach(v => {
    const display = String(v).padStart(2, '0');
    html += `<div class="wheel-item" data-value="${display}">${display}${labelSuffix}</div>`;
  });
  html += '<div class="wheel-spacer"></div>';
  wheel.innerHTML = html;
}

function syncWheelActive(wheel) {
  const items = wheel.querySelectorAll('.wheel-item');
  const index = Math.round(wheel.scrollTop / 40);
  items.forEach((item, i) => {
    item.classList.toggle('active-scroll', i === index);
  });
}

function scrollWheelToValue(id, value) {
  const wheel = document.getElementById(id);
  const items = Array.from(wheel.querySelectorAll('.wheel-item'));
  const targetIndex = items.findIndex(item => item.getAttribute('data-value') === String(value).padStart(2, '0'));
  if (targetIndex !== -1) {
    wheel.scrollTop = targetIndex * 40;
    syncWheelActive(wheel);
  }
}

function changeYear(delta) {
  currentSelectedYear += delta;
  document.getElementById('currentYearDisplay').textContent = currentSelectedYear;
}

function initTimePicker(roundIndex){
  const months = Array.from({length: 12}, (_, i) => i + 1);
  const days = Array.from({length: 31}, (_, i) => i + 1);
  const hours = Array.from({length: 24}, (_, i) => i);
  const minutes = [0, 10, 20, 30, 40, 50];

  populateWheel('wheel-month', months);
  populateWheel('wheel-day', days);
  populateWheel('wheel-hour', hours);
  populateWheel('wheel-minute', minutes);
  
  let targetDate = new Date();
  
  // If editing an existing time, start there
  if (roundTimes[`round_${roundIndex}`]) {
    const parts = roundTimes[`round_${roundIndex}`].match(/(\d+)-(\d+)-(\d+) (\d+):(\d+)/);
    if (parts) {
      targetDate = new Date(parts[1], parts[2]-1, parts[3], parts[4], parts[5]);
    }
  }

  currentSelectedYear = targetDate.getFullYear();
  document.getElementById('currentYearDisplay').textContent = currentSelectedYear;

  const currentMin = targetDate.getMinutes();
  const roundedMin = Math.round(currentMin / 10) * 10 % 60;

  setTimeout(() => {
    scrollWheelToValue('wheel-month', targetDate.getMonth() + 1);
    scrollWheelToValue('wheel-day', targetDate.getDate());
    scrollWheelToValue('wheel-hour', targetDate.getHours());
    scrollWheelToValue('wheel-minute', roundedMin);
  }, 10);
}

function openTimePicker(btn, index){
  activeTimeButton = btn;
  activeRoundIndex = index;
  document.getElementById('timePicker').classList.remove('hidden');
  initTimePicker(index);
}

function closeTimePicker(){
  document.getElementById('timePicker').classList.add('hidden');
}

function confirmTimePicker(){
  const getSelectedValue = (id) => {
    const wheel = document.getElementById(id);
    const index = Math.round(wheel.scrollTop / 40);
    const items = wheel.querySelectorAll('.wheel-item');
    return items[index] ? items[index].getAttribute('data-value') : '00';
  };

  const pad = (n) => String(n).padStart(2,'0');
  const y = currentSelectedYear;
  const m = getSelectedValue('wheel-month');
  const d = getSelectedValue('wheel-day');
  const h = getSelectedValue('wheel-hour');
  const min = getSelectedValue('wheel-minute');
  
  const formattedDate = `${y}-${pad(m)}-${pad(d)} ${pad(h)}:${pad(min)}`;
  
  roundTimes[`round_${activeRoundIndex}`] = formattedDate;

  // Logic: When any round is confirmed, propagate to subsequent rounds
  let baseDateStr = roundTimes[`round_${activeRoundIndex}`];
  if (baseDateStr) {
    const parts = baseDateStr.match(/(\d+)-(\d+)-(\d+) (\d+):(\d+)/);
    let currentDate = new Date(parts[1], parts[2]-1, parts[3], parts[4], parts[5]);
    
    // Propagate to all subsequent rounds
    const totalRounds = parseInt(document.querySelector('.row-input').value || 2);
    for (let i = activeRoundIndex + 1; i < totalRounds; i++) {
      currentDate = new Date(currentDate.getTime() + (24 * 60 * 60 * 1000));
      const y = currentDate.getFullYear();
      const m = pad(currentDate.getMonth() + 1);
      const d = pad(currentDate.getDate());
      const h = pad(currentDate.getHours());
      const min = pad(currentDate.getMinutes());
      roundTimes[`round_${i}`] = `${y}-${m}-${d} ${h}:${min}`;
    }
  }

  // Refresh display
  renderRounds(document.querySelector('.row-input').value);
  closeTimePicker();
}
