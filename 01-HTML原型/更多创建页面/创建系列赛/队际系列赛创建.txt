<script>

let structure = "time";
let mode = "global";
let currentStep = 1;

function nextStep() {
  if (currentStep === 1) {
    document.getElementById('step-config').classList.add('hidden');
    document.getElementById('step-details').classList.remove('hidden');
    document.getElementById('main-btn').innerText = '确定创建';
    document.querySelector('.header-left div:last-child').innerText = '完善赛事信息';
    currentStep = 2;
    window.scrollTo(0,0);
  } else {
    alert('赛事创建成功！');
  }
}

function prevStep() {
  if (currentStep === 2) {
    document.getElementById('step-details').classList.add('hidden');
    document.getElementById('step-config').classList.remove('hidden');
    document.getElementById('main-btn').innerText = '下一步';
    document.querySelector('.header-left div:last-child').innerText = '队际系列赛创建';
    currentStep = 1;
    window.scrollTo(0,0);
  } else {
    // 已经在第一步，可以执行物理返回或其他逻辑
    history.back();
  }
}

function setStructure(type){
  structure = type;
  const card = document.querySelectorAll('.card')[0];
  card.querySelectorAll('.option').forEach(o => o.classList.remove('active'));
  const selectedOption = event.currentTarget;
  selectedOption.classList.add('active');
  selectedOption.querySelector('.ds-radio').checked = true;

  updateLabels();
  render();
}

function updateLabels(){

  if(structure === "space"){
    document.getElementById('labelTitle').innerText = "3. 场地数 S";
    document.getElementById('labelRS').innerText = "S";
  }else{
    document.getElementById('labelTitle').innerText = "3. 轮次数 R";
    document.getElementById('labelRS').innerText = "R";
  }
}

function setMode(m){
  mode = m;
  const isGlobal = mode === 'global';
  const options = document.querySelectorAll('.card')[1].querySelectorAll('.option');
  
  options.forEach((o, i) => {
    const active = (i === 0 && isGlobal) || (i === 1 && !isGlobal);
    o.classList.toggle('active', active);
    const radio = o.querySelector('.ds-radio');
    if (radio) radio.checked = active;
    
    // Manage input visibility
    const existingInput = o.querySelector('input:not([type="radio"])');
    if (i === 0) {
      if (!isGlobal && existingInput) existingInput.remove();
      else if (isGlobal && !existingInput) {
        const input = document.createElement('input');
        input.value = "10";
        input.className = "!mt-0 text-center ml-auto";
        input.onclick = (e) => e.stopPropagation();
        o.appendChild(input);
      }
    } else if (i === 1) {
      if (existingInput) existingInput.remove();
    }
  });

  render();
}

function render(){

  const count = Math.max(2, parseInt(document.getElementById('count').value || 2));
  const container = document.getElementById('container');

  container.innerHTML = '';

  for(let i=1;i<=count;i++){

    const div = document.createElement('div');
    div.className = 'block';

    const title = structure === "space"
      ? `COURSE ${i}`
      : `ROUND ${i}`;

    div.innerHTML = `
      <div class="block-title">${title}</div>

      <div class="option">开球时间</div>
      <div class="option">球场</div>

      <div class="tags">
        <div class="tag active" onclick="selectTag(this)">个人比杆赛</div>
        <div class="tag" onclick="selectTag(this)">四人四球比杆赛</div>
        <div class="tag" onclick="selectTag(this)">四人两球比杆赛</div>
        <div class="tag" onclick="selectTag(this)">最佳球位比杆赛</div>
      </div>

      ${mode === "round" ? `
      <div class="option flex items-center justify-between">
        <span>取本轮次的前</span>
        <div class="flex items-center gap-2">
          <input value="5" class="!mt-0 !w-12 text-center">
          <span class="text-sm">名</span>
        </div>
      </div>
      ` : ``}
    `;

    container.appendChild(div);
  }
}

function setReuse(val) {
  const options = document.querySelectorAll('.card')[3].querySelectorAll('.option');
  options.forEach((o, i) => {
    const active = (i === 0 && val) || (i === 1 && !val);
    o.classList.toggle('active', active);
    const radio = o.querySelector('.ds-radio');
    if (radio) radio.checked = active;
  });
}

function selectTag(el){
  const group = el.parentNode.querySelectorAll('.tag');
  group.forEach(t=>t.classList.remove('active'));
  el.classList.add('active');
}

updateLabels();
render();

</script>
